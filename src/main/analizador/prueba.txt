\\\ 2. Metodo Principal (Regla 618)
Principal {
    \\\ Declaraciones de variables (id-variable inicia en minuscula - Regla 627)
    variable entero contador = 0;
    variable cadena mensaje = "Iniciando calculos";
    variable booleano bandera = verdadero;
    
    imprimir(mensaje);

    \\\ 3. Arreglos y uso de leer() (Reglas 751, 753)
    variable arreglo entero valores [3] = { 10, 20, 30 }; 
    variable entero entrada = leer(); 

    \\\ 4. Estructuras de Control: Condicionales (Regla 717-724)
    si (entrada == 100) {
        imprimir("Valor maximo alcanzado");
    } sino_si (entrada > 50 && entrada < 100) {
        imprimir("Valor aceptable");
    } sino {
        imprimir("Valor muy bajo");
    }

    \\\ 5. Ciclos: Mientras (Regla 733)
    mientras (contador < MAX_INTENTOS) {
        \\\ Llamada a funcion dentro de una operacion (Regla 679)
        variable entero resultado_temp = calcular_area(valores[contador]) + 1;
        
        imprimir(resultado_temp);
        
        contador = contador + 1;
    }

    \\\ 6. Ciclo Para (Regla 728)
    para (variable entero i = 0; i < 3; i = i + 1) {
        si (i == 2) {
            romper;
        }
        imprimir(valores[i]);
    }
}

\\\ 7. Lista de Funciones fuera del principal (Regla 620)
funcion calcular_area (entero radio) {
    \\\ Uso de constante e identificador de variable en operacion
    variable decimal area = PI * (radio * radio);
    retornar area;
}

funcion finalizar_sistema () {
    imprimir("Programa terminado con exito");
    retornar;
}
