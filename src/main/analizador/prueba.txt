funcion factorial(entero n){
    variable entero resultado = 1;
    para(variable entero i = 1; i <= n; i =+ 1){
        resultado =* i;
    }
    retornar resultado;
}

funcion sumaHasta(entero n){
    variable entero suma = 0;
    para(variable entero i = 1; i <= n; i =+ 1){
        suma =+ i;
    }
    retornar suma;
}

funcion tablaMultiplicar(entero n){
